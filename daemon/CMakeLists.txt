cmake_minimum_required(VERSION 3.25)
file(STRINGS "${CMAKE_CURRENT_LIST_DIR}/../VERSION" DASHBOARD_DAEMON_VERSION LIMIT_COUNT 1)
if(NOT DASHBOARD_DAEMON_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
  message(FATAL_ERROR "VERSION must contain one stable MAJOR.MINOR.PATCH version")
endif()
project(DashboardDaemon VERSION ${DASHBOARD_DAEMON_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(dashboard_daemon_core STATIC src/daemon.cpp)
target_include_directories(dashboard_daemon_core PUBLIC include)
target_compile_definitions(dashboard_daemon_core PUBLIC DASHBOARD_DAEMON_VERSION="${PROJECT_VERSION}")
add_executable(dashboard-daemon src/main.cpp)
target_link_libraries(dashboard-daemon PRIVATE dashboard_daemon_core)
# The daemon is also built by a pinned musl release toolchain whose diagnostics are
# intentionally independent from the Qt application's clang-tidy wrapper.
set_target_properties(dashboard_daemon_core dashboard-daemon PROPERTIES CXX_CLANG_TIDY "")
if(BUILD_TESTING)
  add_executable(dashboard_daemon_test tests/daemon_test.cpp)
  target_link_libraries(dashboard_daemon_test PRIVATE dashboard_daemon_core)
  set_target_properties(dashboard_daemon_test PROPERTIES CXX_CLANG_TIDY "")
  add_test(NAME dashboard_daemon_test COMMAND dashboard_daemon_test)
  add_test(NAME dashboard_daemon_install_test COMMAND sh ${CMAKE_CURRENT_SOURCE_DIR}/tests/install_test.sh ${CMAKE_CURRENT_SOURCE_DIR}/package/install.sh)
endif()
install(TARGETS dashboard-daemon RUNTIME DESTINATION bin)
